cmake_minimum_required(VERSION 3.28)
project(qcopilots-search VERSION 0.1.0 LANGUAGES C CXX)

file(REAL_PATH "${CMAKE_SOURCE_DIR}" _qcopilots_source)
file(REAL_PATH "${CMAKE_BINARY_DIR}" _qcopilots_binary)
cmake_path(IS_PREFIX _qcopilots_source "${_qcopilots_binary}" NORMALIZE _qcopilots_in_source)
if(_qcopilots_in_source)
  message(FATAL_ERROR "Use an external build directory. Run qcopilots-search-dev configure --build-root <directory>.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.4" CACHE STRING "Minimum supported macOS version")
  set(CMAKE_INSTALL_RPATH "@executable_path/../Frameworks")
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(MSVC)
  add_compile_options(/utf-8 /permissive- /Zc:__cplusplus)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")
include(GNUInstallDirs)
if(APPLE)
  # A user drags one .app out of the DMG. All helpers, libraries and data must
  # travel with that bundle, rather than rely on adjacent staging directories.
  set(CMAKE_INSTALL_BINDIR "qcopilots-search.app/Contents/MacOS")
  set(CMAKE_INSTALL_LIBDIR "qcopilots-search.app/Contents/Frameworks")
  set(CMAKE_INSTALL_DATADIR "qcopilots-search.app/Contents/Resources")
endif()
include(CTest)
# CTest defaults to only 1 KiB for passing tests, which removes QtTest's case
# totals and can hide internal skips from the release evidence.
file(WRITE "${CMAKE_BINARY_DIR}/CTestCustom.cmake"
  "set(CTEST_CUSTOM_MAXIMUM_PASSED_TEST_OUTPUT_SIZE 16777216)\nset(CTEST_CUSTOM_MAXIMUM_FAILED_TEST_OUTPUT_SIZE 16777216)\n")
option(QCOPILOTS_SEARCH_BUILD_TESTS "Build application tests" ON)
option(QCOPILOTS_SEARCH_DEPLOY_QT "Deploy Qt dependencies during installation" ON)
option(QCOPILOTS_SEARCH_REQUIRE_FULL_FEATURES "Require every extraction backend" ON)
set(QCOPILOTS_SEARCH_DEPENDENCY_ROOT "${CMAKE_BINARY_DIR}/../dependencies" CACHE PATH "External dependency source/cache root")
set(QCOPILOTS_SEARCH_TEST_ROOT "${CMAKE_BINARY_DIR}/test-runs" CACHE PATH "External root for isolated test runs")
set(QCOPILOTS_SEARCH_MODEL_DIR "" CACHE PATH "Optional read-only model cache for real model tests")
set(FETCHCONTENT_BASE_DIR "${QCOPILOTS_SEARCH_DEPENDENCY_ROOT}/fetch" CACHE PATH "Dependency working directory")
file(MAKE_DIRECTORY "${QCOPILOTS_SEARCH_TEST_ROOT}")

find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets Network Sql Concurrent Test Xml Svg Core5Compat Multimedia MultimediaWidgets Pdf PdfWidgets)
if(NOT Qt6Core_VERSION VERSION_EQUAL "6.12.0")
  message(FATAL_ERROR "The locked SDK requires Qt Core 6.12.0, found ${Qt6Core_VERSION}. Run bootstrap with the selected Qt root.")
endif()
if(NOT Qt6Pdf_VERSION VERSION_EQUAL "6.140.0")
  message(FATAL_ERROR "The locked independent Qt PDF module requires 6.140.0, found ${Qt6Pdf_VERSION}. Run bootstrap with the selected Qt root.")
endif()
qt_standard_project_setup(REQUIRES 6.12)
include_directories("${PROJECT_SOURCE_DIR}/src")
set(CMAKE_AUTOMOC OFF)
set(CMAKE_AUTORCC OFF)
set(CMAKE_AUTOUIC OFF)
include(QCopilotsSearchNativeDependencies)
qcopilots_search_add_native_dependencies()
include(QCopilotsSearchFFmpeg)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

foreach(_module core runtime extraction storage service app cli worker)
  if(EXISTS "${PROJECT_SOURCE_DIR}/src/${_module}/CMakeLists.txt")
    add_subdirectory("src/${_module}")
  endif()
endforeach()
include(QCopilotsSearchSymbols)
if(QCOPILOTS_SEARCH_BUILD_TESTS AND EXISTS "${PROJECT_SOURCE_DIR}/tests/CMakeLists.txt")
  enable_testing()
  add_subdirectory(tests)
endif()
if(QCOPILOTS_SEARCH_BUILD_TESTS)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME qcopilots-search-tests-workflow COMMAND "${Python3_EXECUTABLE}"
    "${PROJECT_SOURCE_DIR}/tools/qcopilots-search-test-workflow.py")
  set_tests_properties(qcopilots-search-tests-workflow PROPERTIES
    WORKING_DIRECTORY "${CMAKE_BINARY_DIR}" LABELS "unit;portability;workflow"
    ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()
include(QCopilotsSearchDevelopment)
include(QCopilotsSearchPackaging)
