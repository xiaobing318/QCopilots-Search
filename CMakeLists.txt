cmake_minimum_required(VERSION 3.28)
project(qcs VERSION 0.1.0 LANGUAGES C CXX)

file(REAL_PATH "${CMAKE_SOURCE_DIR}" _qcopilots_source)
file(REAL_PATH "${CMAKE_BINARY_DIR}" _qcopilots_binary)
cmake_path(IS_PREFIX _qcopilots_source "${_qcopilots_binary}" NORMALIZE _qcopilots_in_source)
if(_qcopilots_in_source)
  message(FATAL_ERROR "Use an external build directory. Run qcs-dev configure --build-root <directory>.")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  set(_qcs_object_path_max 200)
else()
  set(_qcs_object_path_max 240)
endif()
set(CMAKE_OBJECT_PATH_MAX ${_qcs_object_path_max} CACHE STRING "Maximum generated object path length")
if(NOT CMAKE_OBJECT_PATH_MAX MATCHES "^[0-9]+$" OR CMAKE_OBJECT_PATH_MAX LESS 128)
  message(FATAL_ERROR "CMAKE_OBJECT_PATH_MAX must be an integer of at least 128")
endif()
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.4" CACHE STRING "Minimum supported macOS version")
  set(CMAKE_INSTALL_RPATH "@executable_path/../Frameworks")
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(MSVC)
  add_compile_options(/utf-8 /permissive- /Zc:__cplusplus)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")
include(qcs-output-paths)
include(qcs-msvc-rsp)
include(GNUInstallDirs)
if(APPLE)
  # A user drags one .app out of the DMG. All helpers, libraries and data must
  # travel with that bundle, rather than rely on adjacent staging directories.
  set(CMAKE_INSTALL_BINDIR "qcs-gui.app/Contents/MacOS")
  set(CMAKE_INSTALL_LIBDIR "qcs-gui.app/Contents/Frameworks")
  set(CMAKE_INSTALL_DATADIR "qcs-gui.app/Contents/Resources")
endif()
# --prefix and DESTDIR may change after configure. Check their effective path
# before any subdirectory installation can copy files into the source tree.
qcopilots_search_install_output_guard("${CMAKE_SOURCE_DIR}")
include(CTest)
# CTest defaults to only 1 KiB for passing tests, which removes QtTest's case
# totals and can hide internal skips from the release evidence.
file(WRITE "${CMAKE_BINARY_DIR}/CTestCustom.cmake"
  "set(CTEST_CUSTOM_MAXIMUM_PASSED_TEST_OUTPUT_SIZE 16777216)\nset(CTEST_CUSTOM_MAXIMUM_FAILED_TEST_OUTPUT_SIZE 16777216)\n")
option(QCOPILOTS_SEARCH_BUILD_TESTS "Build application tests" ON)
option(QCOPILOTS_SEARCH_DEPLOY_QT "Deploy Qt dependencies during installation" ON)
option(QCOPILOTS_SEARCH_REQUIRE_FULL_FEATURES "Require every extraction backend" ON)
set(QCOPILOTS_SEARCH_DEPENDENCY_ROOT "${CMAKE_BINARY_DIR}/../deps" CACHE PATH "External dependency source/cache root")
set(QCOPILOTS_SEARCH_TEST_ROOT "${CMAKE_BINARY_DIR}/../runs" CACHE PATH "External root for isolated test runs")
set(QCOPILOTS_SEARCH_MODEL_DIR "" CACHE PATH "Optional read-only model cache for real model tests")
set(FETCHCONTENT_BASE_DIR "${QCOPILOTS_SEARCH_DEPENDENCY_ROOT}" CACHE PATH "Dependency working directory")
foreach(_root QCOPILOTS_SEARCH_DEPENDENCY_ROOT QCOPILOTS_SEARCH_TEST_ROOT FETCHCONTENT_BASE_DIR)
  qcopilots_search_require_external_path("${_root}" "${${_root}}")
endforeach()
get_cmake_property(_qcopilots_variables VARIABLES)
foreach(_variable IN LISTS _qcopilots_variables)
  if(_variable MATCHES "^CMAKE_(RUNTIME|LIBRARY|ARCHIVE|PDB|COMPILE_PDB)_OUTPUT_DIRECTORY(_[A-Za-z0-9_]+)?$"
      OR _variable MATCHES "^FETCHCONTENT_SOURCE_DIR_")
    if(NOT "${${_variable}}" STREQUAL "")
      qcopilots_search_require_external_path("${_variable}" "${${_variable}}")
    endif()
  endif()
endforeach()
file(MAKE_DIRECTORY "${QCOPILOTS_SEARCH_TEST_ROOT}")

  message(FATAL_ERROR "The locked SDK requires Qt Core 6.12.0, found ${Qt6Core_VERSION}. Run bootstrap with the selected Qt root.")
endif()
if(NOT Qt6Pdf_VERSION VERSION_EQUAL "6.140.0")
  message(FATAL_ERROR "The locked independent Qt PDF module requires 6.140.0, found ${Qt6Pdf_VERSION}. Run bootstrap with the selected Qt root.")
endif()

  enable_testing()

  set_tests_properties(qcs-tests-release-tool PROPERTIES

    ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
