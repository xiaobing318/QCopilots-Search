cmake_minimum_required(VERSION 3.28)
project(qcs-ocr-fixture-generator LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui)
add_executable(qcs-generate-ocr-fixtures qcs-generate-ocr-fixtures.cpp)
target_compile_features(qcs-generate-ocr-fixtures PRIVATE cxx_std_20)
target_link_libraries(qcs-generate-ocr-fixtures PRIVATE Qt6::Core Qt6::Gui)
if(MSVC)
    target_compile_options(qcs-generate-ocr-fixtures PRIVATE /utf-8)
endif()
